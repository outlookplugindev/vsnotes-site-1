🧠 Build the Semantic Kernel
Dependency Injection · OpenAI Connector · Kernel Builder
 Program.cs — Kernel Setup
using Microsoft.SemanticKernel;
using Microsoft.SemanticKernel.Connectors.OpenAI;

var builder = WebApplication.CreateBuilder(args);
var config = builder.Configuration;

// Build Semantic Kernel with OpenAI
var kernel = Kernel.CreateBuilder()
.AddOpenAIChatCompletion(
modelId: config["OpenAI:Model"]!,
apiKey: config["OpenAI:ApiKey"]!
)
.AddOpenAITextEmbeddingGeneration(
modelId: config["OpenAI:EmbeddingModel"]!,
apiKey: config["OpenAI:ApiKey"]!
)
.Build();

// Register kernel as singleton
builder.Services.AddSingleton(kernel);
💡 For Azure OpenAI, replace AddOpenAIChatCompletion with AddAzureOpenAIChatCompletion(deploymentName, endpoint, apiKey).

📄 Document Loading & Chunking
PdfPig · TextChunker · Memory Ingestion

Use PdfPig to read PDFs and Semantic Kernel's TextChunker to split them.

 DocumentLoader.cs
using UglyToad.PdfPig;
using Microsoft.SemanticKernel.Text;

public static class DocumentLoader
{
public static List<string> LoadAndChunkPdf(string path)
{
var text = new StringBuilder();

// Extract text from PDF using PdfPig
using var doc = PdfDocument.Open(path);
foreach var page in doc.GetPages()
text.AppendLine(page.Text);

// Split into overlapping chunks
var chunks = TextChunker.SplitPlainTextLines(
text.ToString(),
maxTokensPerLine: 100
);

return TextChunker.SplitPlainTextParagraphs(
chunks, maxTokensPerParagraph: 300,
overlapTokens: 50
);
}
}
4
💾 Semantic Memory (Vector Store)
VolatileMemoryStore · Embeddings · SaveAsync
 MemoryService.cs
using Microsoft.SemanticKernel.Memory;
using Microsoft.SemanticKernel.Connectors.OpenAI;

public class MemoryService
{
private readonly ISemanticTextMemory _memory;
private const string Collection = "documents";

public MemoryService(string apiKey, string embeddingModel)
{
_memory = new MemoryBuilder()
.WithOpenAITextEmbeddingGeneration(embeddingModel, apiKey)
.WithMemoryStore(new VolatileMemoryStore())
.Build();
}

public async Task IngestDocumentsAsync(List<string> chunks)
{
for int i = 0; i < chunks.Count; i++
{
await _memory.SaveInformationAsync(
collection: Collection,
text: chunks[i],
id: $"chunk-{i}"
);
}
Console.WriteLine($"✅ Ingested {chunks.Count} chunks");
}

public async Task<string> SearchAsync(string query, int topK = 4)
{
var results = await _memory
.SearchAsync(Collection, query, limit: topK)
.ToListAsync();

return string.Join("\n\n", results.Select(r => r.Metadata.Text));
}
}
📌 For production, replace VolatileMemoryStore with QdrantMemoryStore or AzureAISearchMemoryStore for persistent, scalable vector storage.
5
🔌 RAG Kernel Plugin
KernelFunction · SKFunction Attribute · Plugin Registration

In Semantic Kernel, Plugins are collections of KernelFunctions that the AI can call. Create a RAG plugin:

 RagPlugin.cs
using Microsoft.SemanticKernel;
using System.ComponentModel;

public class RagPlugin
{
private readonly MemoryService _memoryService;

public RagPlugin(MemoryService memoryService)
=> _memoryService = memoryService;

[KernelFunction]
[Description("Search the internal knowledge base for relevant information")]
public async Task<string> SearchDocumentsAsync(
[Description("The search query")] string query)
{
var context = await _memoryService.SearchAsync(query);
return string.IsNullOrEmpty(context)
? "No relevant information found in the knowledge base."
: $"Relevant context:\n{context}";
}
}

// Register plugin with kernel
kernel.Plugins.AddFromObject(new RagPlugin(memoryService), "RAGPlugin");

⚡ Agentic Planner
FunctionCallingStepwisePlanner · Multi-step Reasoning

Semantic Kernel's FunctionCallingStepwisePlanner enables the agent to reason step-by-step and call plugins autonomously.

 AgentService.cs
using Microsoft.SemanticKernel;
using Microsoft.SemanticKernel.Planning;

public class AgentService
{
private readonly Kernel _kernel;

public AgentService(Kernel kernel) => _kernel = kernel;

public async Task<string> AskAsync(string question)
{
// Create stepwise planner
var planner = new FunctionCallingStepwisePlanner(
new FunctionCallingStepwisePlannerOptions
{
MaxIterations = 10,
MaxTokens = 4000
}
);

// Execute plan
var result = await planner.ExecuteAsync(_kernel, question);
return result.FinalAnswer ?? "I could not find an answer.";
}
}
💡 The planner will automatically call SearchDocumentsAsync from our RAGPlugin when it needs to retrieve information — no extra code needed!

🌐 ASP.NET Core Web API
Minimal API · REST Endpoint · Swagger UI
 Program.cs — Complete API
using Microsoft.SemanticKernel;

var builder = WebApplication.CreateBuilder(args);
var config = builder.Configuration;

// 1. Build Kernel
var kernel = Kernel.CreateBuilder()
.AddOpenAIChatCompletion(config["OpenAI:Model"]!, config["OpenAI:ApiKey"]!)
.AddOpenAITextEmbeddingGeneration(config["OpenAI:EmbeddingModel"]!, config["OpenAI:ApiKey"]!)
.Build();

// 2. Setup memory and ingest documents
var memSvc = new MemoryService(config["OpenAI:ApiKey"]!, config["OpenAI:EmbeddingModel"]!);
var chunks = DocumentLoader.LoadAndChunkPdf("./docs/knowledge.pdf");
await memSvc.IngestDocumentsAsync(chunks);

// 3. Register RAG plugin
kernel.Plugins.AddFromObject(new RagPlugin(memSvc), "RAGPlugin");

// 4. Register services
builder.Services.AddSingleton(kernel);
builder.Services.AddSingleton<AgentService>();
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();
app.UseSwagger(); app.UseSwaggerUI();

// 5. Define endpoint
app.MapPost("/ask", async (QuestionRequest req, AgentService agent) =>
{
var answer = await agent.AskAsync(req.Question);
return Results.Ok(new { answer });
}).WithOpenApi();

app.Run();

record QuestionRequest(string Question);

Test with Swagger UI at https://localhost:5001/swagger or call via POST /ask.
